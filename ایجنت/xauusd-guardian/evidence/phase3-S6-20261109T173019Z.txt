"""S6 live demonstration: additive apply, reversal, pre-0003 preservation, re-apply.

    C:\\Python311\\python.exe C:\\GS5\\s6_demo.py

from __future__ import annotations

import shutil
import sys
import tempfile
from pathlib import Path

ROOT = Path(r"C:\GS5\repo") / "ایجنت" / "xauusd-guardian"
sys.path.insert(0, str(ROOT / "src"))

from guardian.storage.db import (  # noqa: E402
    MigrationError,
    apply_migrations,
    connect,
    migration_history,
    reversal_plan,
    revert_migration,
)

MIGRATIONS = ROOT / "migrations"
START = "2026-11-05T08:00:00+00:00"


def tables(conn) -> list[str]:
    return sorted(row["name"] for row in conn.execute("SELECT name FROM sqlite_master WHERE type='table'"))


def ledger(conn) -> list[int]:
    return [entry.version for entry in migration_history(conn)]


def counted(conn, table: str) -> int:
    return conn.execute(f"SELECT COUNT(*) AS c FROM {table}").fetchone()["c"]


def main() -> int:
    work = Path(tempfile.mkdtemp(prefix="s6-demo-"))
    db = work / "demo.sqlite3"
    legacy = work / "legacy"
    legacy.mkdir()
    for name in ("0001_init.sql", "0002_market_data.sql"):
        shutil.copy(MIGRATIONS / name, legacy / name)

    conn = connect(db)

    print("1) apply 0001+0002 ->", apply_migrations(conn, legacy))
    for run_id, close, bar_time in (("run-a", 1.5, START), ("run-b", 2.5, "2026-11-05T08:05:00+00:00")):
        conn.execute(
            "INSERT INTO feed_runs (run_id, symbol, kind, started_utc, status) VALUES (?,?,?,?,?)",
            (run_id, "XAUUSD.", "bars", START, "OK"),
        )
        conn.execute(
            "INSERT INTO bars (symbol, timeframe, time_utc, open, high, low, close, tick_volume, "
            "spread_points, real_volume, content_sha256, first_seen_utc, last_seen_utc, run_id) "
            "VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?)",
            ("XAUUSD.", "M5", bar_time, 1.0, 2.0, 0.5, close, 10, 0, 0.0, "sha-" + run_id, START, START, run_id),
        )
    conn.commit()
    pre_bars, pre_runs = counted(conn, "bars"), counted(conn, "feed_runs")
    print(f"   seeded pre-0003 rows: bars={pre_bars} feed_runs={pre_runs}")

    print("2) apply 0003 ->", apply_migrations(conn, MIGRATIONS))
    print("   signals+signal_runs present:", {"signals", "signal_runs"} <= set(tables(conn)))
    conn.execute(
        "INSERT INTO signal_runs (run_id, symbol, strategy_id, timeframe, from_utc, to_utc, bars_seen, "
        "signals, written, identical, status, created_utc) VALUES (?,?,?,?,?,?,?,?,?,?,?,?)",
        ("run_s6", "XAUUSD.", "ema_cross", "M5", START, START, 2, 1, 1, 0, "OK", START),
    )
    conn.execute(
        "INSERT INTO signals (symbol, timeframe, bar_time_utc, strategy_id, direction, strength, "
        "reason_code, run_id, created_utc) VALUES (?,?,?,?,?,?,?,?,?)",
        ("XAUUSD.", "M5", START, "ema_cross", "LONG", "0.5", "EMA_CROSS", "run_s6", START),
    )
    conn.commit()
    print(f"   post-0003 rows: signals={counted(conn, 'signals')} signal_runs={counted(conn, 'signal_runs')}")
    print("   ledger:", ledger(conn))

    plan = reversal_plan(conn, MIGRATIONS, 3)
    print(f"3) reversal plan: version={plan.version} down={plan.down_filename} sha256={plan.down_sha256}")
    print(f"   tables to drop: {list(plan.tables)}  indexes: {list(plan.indexes)}")

    print("4) revert 0003 ->", revert_migration(conn, MIGRATIONS, 3))
    after = set(tables(conn))
    print("   0003 objects still present:", sorted({"signals", "signal_runs", "idx_signals_run"} & after))
    print(f"   pre-0003 rows intact: bars={counted(conn, 'bars')} (was {pre_bars}) "
          f"feed_runs={counted(conn, 'feed_runs')} (was {pre_runs})")
    print("   ledger:", ledger(conn))

    print("5) re-apply 0003 ->", apply_migrations(conn, MIGRATIONS))
    print("   tables back, rows in signals =", counted(conn, "signals"))
    print(f"   bars still {counted(conn, 'bars')}  ledger: {ledger(conn)}")
    print("   second apply is a no-op ->", apply_migrations(conn, MIGRATIONS))

    hostile = work / "hostile"
    shutil.copytree(MIGRATIONS, hostile)
    (hostile / "rollback" / "0003_signals.down.sql").write_text(
        "DROP TABLE signals;\nDROP TABLE signal_runs;\nDROP TABLE bars;\n", encoding="utf-8"
    )
    try:
        revert_migration(conn, hostile, 3)
        print("6) guard: FAILED - a reverse reached a pre-0003 table")
        return 1
    except MigrationError as exc:
        print("6) guard refused a reverse that would drop a pre-0003 table:")
        print("   MigrationError:", exc)
    print(f"   bars still {counted(conn, 'bars')}  ledger: {ledger(conn)}")

    print("RESULT: additive + reversible + pre-0003 data preserved + deterministic re-apply")
    conn.close()
    shutil.rmtree(work, ignore_errors=True)
    return 0


if __name__ == "__main__":
    sys.exit(main())
